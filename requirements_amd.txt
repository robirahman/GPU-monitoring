#   1. Install ROCm 6.4+ (RDNA 4 needs >=6.4; ROCm 7.x preferred). The amdsmi
#      Python bindings ship with ROCm at /opt/rocm/share/amd_smi and are
#      imported via PYTHONPATH, NOT via pip:
#          export PYTHONPATH=/opt/rocm/share/amd_smi:$PYTHONPATH
#      The collector adds this path automatically.
#   2. Install PyTorch with the ROCm wheel (replace ROCM_VER with the
#      version matching your ROCm install — see https://pytorch.org/):
#          pip install --index-url https://download.pytorch.org/whl/rocmROCM_VER \
#              torch torchvision torchaudio
#   3. (Optional, for Tier 2) Install RDC and start its daemon:
#          sudo apt install rdc
#          sudo systemctl enable --now rdc
#
