% --- Float containment -------------------------------------------------
% Floats default to drifting into whatever page has room, which pushed four
% figures out of the section that introduces them. A FloatBarrier before every
% section that needs one forces LaTeX to flush pending floats first, so each
% figure and table appears under its own heading. The barriers are placed in
% the source (body_main.tex, body_appendix.tex) rather than by patching
% \section, so the main text pays for only the one it actually needs.
\usepackage{float}
\usepackage{placeins}
% Barriers alone would push any float that cannot fit the remaining space onto
% a page of its own. Loosening the default area fractions lets a large float
% share a page with text instead, which keeps the page count down.
\renewcommand{\topfraction}{0.92}
\renewcommand{\bottomfraction}{0.85}
\renewcommand{\textfraction}{0.06}
\renewcommand{\floatpagefraction}{0.80}
\setcounter{topnumber}{3}
\setcounter{bottomnumber}{2}
\setcounter{totalnumber}{5}

% --- Author footnote marks --------------------------------------------
% neurips_2026.sty sets \@makefnmark to \hbox to \z@{...}, a zero-width box, so
% that marker glyphs do not shift the centred author names. Two \thanks on one
% author then print on top of each other. This is the style's own \maketitle
% with that one line changed to give the in-text mark its natural width. The
% footnote text at the bottom of the page is unaffected: \@makefntext below
% typesets the symbol itself rather than calling \@makefnmark. Redefined in
% full rather than patched because the original contains a parameter token,
% which \patchcmd cannot handle.
\makeatletter
\renewcommand{\maketitle}{%
  \par
  \begingroup
    \renewcommand{\thefootnote}{\fnsymbol{footnote}}
    \renewcommand{\@makefnmark}{\textsuperscript{\@thefnmark}}% <- was zero-width
    \long\def\@makefntext##1{%
      \parindent 1em\noindent
      \hbox to 1.8em{\hss $\m@th ^{\@thefnmark}$}##1
    }
    \thispagestyle{empty}
    \@maketitle
    \@thanks
    \@notice
  \endgroup
  \let\maketitle\relax
  \let\thanks\relax
}
\makeatother
